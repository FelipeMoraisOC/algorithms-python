print('Verificar se numero é triangular')  
num = int(input('Digite o número: ')) 
i = 1 
while i < num : 
    tri = i * (i+1) * (i+2) 
    if tri == num: 
        print('numero é triangular') 
        break 
    elif tri > num: 
        print('numero não é triangular') 
        break 
    i += 1

print('determine a sua decomposição em fatores primos calculando também a multiplicidade de cada fator') 
def ver_primo(n: int) -> bool: 
    i = 1 
    j = 1 
    primo = True 
    for i in range(n): 
        for j in range(n): 
            aux = i * j 
            if aux == n and i != 1 and j != 1: 
                primo = False 
                break 
            if aux > n: break 
        if not primo: break 
    return primo
